1. 图片格式与存储规范：
使用WebP格式存储图片，它具有更好的压缩率和质量平衡
为每种图片类型（regular、animated等）创建不同的尺寸版本：
缩略图：200x200
标准图：400x400
高清图：800x800
文件命名规则：{dessertId}_{style}_{size}.webp
例如：chocolate_cake_regular_400.webp

   class ImageCache {
       static let shared = ImageCache()
       private let cache = NSCache<NSString, UIImage>()
       private let fileManager = FileManager.default
       
       // 内存缓存
       func setImage(_ image: UIImage, forKey key: String) {
           cache.setObject(image, forKey: key as NSString)
       }
       
       // 本地持久化缓存
       func saveImageToLocal(_ image: UIImage, forKey key: String) {
           guard let data = image.jpegData(compressionQuality: 0.8) else { return }
           let path = getLocalPath(for: key)
           try? data.write(to: path)
       }
   }
4 图片加载优化：
   class ImageLoader {
       static let shared = ImageLoader()
       private let cache = ImageCache.shared
       
       func loadImage(for dessert: DessertItem, style: FoodImageStyle, completion: @escaping (UIImage?) -> Void) {
           // 1. 检查内存缓存
           if let cachedImage = cache.getImage(forKey: dessert.getFullImageName(for: style)) {
               completion(cachedImage)
               return
           }
           
           // 2. 检查本地缓存
           if let localImage = cache.getLocalImage(forKey: dessert.getFullImageName(for: style)) {
               completion(localImage)
               return
           }
           
           // 3. 从云端加载
           loadFromCloud(dessert: dessert, style: style) { image in
               if let image = image {
                   self.cache.setImage(image, forKey: dessert.getFullImageName(for: style))
                   self.cache.saveImageToLocal(image, forKey: dessert.getFullImageName(for: style))
               }
               completion(image)
           }
       }
   }
